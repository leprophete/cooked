  function hashString(str) {

    let h1 = 0xdeadbeef, h2 = 0x41c6ce57;

    for (let i = 0; i < str.length; i++) {

      const c = str.charCodeAt(i);

      h1 = Math.imul(h1 ^ c, 2654435761);

      h2 = Math.imul(h2 ^ c, 1597334677);

    }

    h1 = Math.imul(h1 ^ (h1 >>> 16), 2246822507) ^ Math.imul(h2 ^ (h2 >>> 13), 3266489909);

    return h1 >>> 0;

  }

  function seededRandom(seed) {

    return function () {

      seed |= 0; seed = (seed + 0x6d2b79f5) | 0;

      let t = Math.imul(seed ^ (seed >>> 15), 1 | seed);

      t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;

      return ((t ^ (t >>> 14)) >>> 0) / 4294967296;

    };

  }

  function countLines(text, maxW) {

  function wrapText(text, x, y, maxW, lineH, maxLines) {

  function parseInput(raw) {

    const v = raw.trim().replace(/^https?:\/\/(www\.)?(x|twitter)\.com\//i, '').replace(/\/.*$/, '');

    if (isSolanaAddress(v)) return { type: 'wallet', value: v };

  }

  function todayKey() { return new Date().toISOString().slice(0, 10); }


  // Each day has unlimited rounds; every round is a fixed pair of memes and fighters.

  function getBattle() {

    const day = todayKey();

    let saved = store.get('cooked_battle', null);

    if (!saved || saved.day !== day || !saved.rounds) saved = { day, round: 0, rounds: {} };

    const round = saved.round;

    const rand = seededRandom(hashString('battle:' + day + ':' + round));

    const i = Math.floor(rand() * MEMES.length);

    let j = Math.floor(rand() * (MEMES.length - 1));

    if (j >= i) j++;

    let fighterA = makeFighter(rand), fighterB = makeFighter(rand);

    while (fighterB.label === fighterA.label) fighterB = makeFighter(rand);

    const base = { a: 400 + Math.floor(rand() * 900), b: 400 + Math.floor(rand() * 900) };

    const state = saved.rounds[round] || { a: 0, b: 0, voted: null };

    return { saved, round, memeA: MEMES[i], memeB: MEMES[j], fighterA, fighterB, base, state };

  }

    const now = new Date();

    const end = new Date(now); end.setUTCHours(24, 0, 0, 0);

    const s = Math.max(0, Math.floor((end - now) / 1000));

    const hh = String(Math.floor(s / 3600)).padStart(2, '0');

    const mm = String(Math.floor((s % 3600) / 60)).padStart(2, '0');

    const ss = String(s % 60).padStart(2, '0');

    $('battleTimer').textContent = `${hh}:${mm}:${ss}`;

    if (s === 0) renderBattle();

  }
